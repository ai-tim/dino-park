let audioCtx = null; function playDinoRoar(isPlant) { try { if (!audioCtx) { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } if (audioCtx.state === 'suspended') { audioCtx.resume(); } const startTime = audioCtx.currentTime; const duration = isPlant ? 0.8 : 1.4; // Основной генератор баса const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain(); // Фильтр низких частот (срезает писк, оставляя плотный гул и рык) const filter = audioCtx.createBiquadFilter(); filter.type = 'lowpass'; filter.frequency.setValueAtTime(isPlant ? 300 : 150, startTime); filter.frequency.exponentialRampToValueAtTime(60, startTime + duration); osc.type = isPlant ? 'triangle' : 'sawtooth'; // Утробные низкие частоты osc.frequency.setValueAtTime(isPlant ? 120 : 70, startTime); osc.frequency.exponentialRampToValueAtTime(isPlant ? 40 : 20, startTime + duration); // Вибрато для дрожания голоса const modulator = audioCtx.createOscillator(); const modGain = audioCtx.createGain(); modulator.frequency.value = 14; modGain.gain.value = 25; modulator.connect(osc.frequency); modulator.start(startTime); modulator.stop(startTime + duration); // Огибающая громкости gain.gain.setValueAtTime(0.01, startTime); gain.gain.linearRampToValueAtTime(0.5, startTime + 0.2); gain.gain.exponentialRampToValueAtTime(0.001, startTime + duration); // Соединяем цепочку: Генератор -> Фильтр -> Громкость -> Динамик osc.connect(filter); filter.connect(gain); gain.connect(audioCtx.destination); osc.start(startTime); osc.stop(startTime + duration); } catch (e) { console.log("Ошибка аудио:", e); } }
